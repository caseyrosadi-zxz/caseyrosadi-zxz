/* ============ LOGIKA QUIZ ============ */
const $ = id => document.getElementById(id);
let idx = 0;
let jawab = new Array(SOAL.length).fill(null);

$("start").onclick = () => { $("intro").classList.add("hidden"); $("quiz").classList.remove("hidden"); show(); };
$("back").onclick = () => { if (idx > 0) { idx--; show(); } };
$("next").onclick = () => { if (idx < SOAL.length - 1) { idx++; show(); } else hasil(); };

function show(){
  const s = SOAL[idx];
  $("q").textContent = s.q;
  $("count").textContent = "Pertanyaan " + (idx+1) + " dari " + SOAL.length;
  $("prog").style.width = (idx / SOAL.length * 100) + "%";
  $("opts").innerHTML = "";
  s.o.forEach((o, i) => {
    const b = document.createElement("button");
    b.className = "opt" + (jawab[idx] === i ? " sel" : "");
    b.textContent = o[0];
    b.onclick = () => { jawab[idx] = i; show(); };
    $("opts").appendChild(b);
  });
  $("next").disabled = jawab[idx] === null;
  $("next").textContent = idx === SOAL.length - 1 ? "Lihat hasil" : "Lanjut";
  $("back").style.visibility = idx === 0 ? "hidden" : "visible";
}

function hasil(){
  // Hitung skor tiap kategori
  const skor = {}, maks = {};
  Object.keys(KATEGORI).forEach(k => { skor[k] = 0; maks[k] = 0; });
  SOAL.forEach((s, n) => {
    s.o.forEach(o => { for (const k in o[1]) maks[k] = Math.max(maks[k], 0); });
    // skor maksimum per soal untuk tiap kategori
    const best = {};
    s.o.forEach(o => { for (const k in o[1]) best[k] = Math.max(best[k] || 0, o[1][k]); });
    for (const k in best) maks[k] += best[k];
    const pilih = s.o[jawab[n]][1];
    for (const k in pilih) skor[k] += pilih[k];
  });

  // Skor tiap UKM sebagai persentase kecocokan
  const urut = UKM.map(u => ({ ...u, pct: Math.round(skor[u.cat] / (maks[u.cat] || 1) * 100) }))
                  .sort((a, b) => b.pct - a.pct);
  const top = urut[0], lain = urut.slice(1, 4);

  $("quiz").classList.add("hidden");
  const r = $("result");
  r.classList.remove("hidden");
  r.innerHTML = `
    <div class="card top">
      <span class="badge">${KATEGORI[top.cat]}</span>
      <p class="muted" style="margin-bottom:4px">UKM yang paling cocok untukmu</p>
      <h1>${top.nama}</h1>
      <p><span class="pct">${top.pct}% cocok.</span> ${top.desc}</p>
    </div>
    <div class="card" style="margin-top:16px">
      <h2 style="font-size:1.15rem">Pilihan lain yang layak dicoba</h2>
      ${lain.map(u => `<div class="alt"><div><strong>${u.nama}</strong><br><span class="muted">${u.desc}</span></div><span class="pct">${u.pct}%</span></div>`).join("")}
    </div>
    <div style="margin-top:18px"><button class="btn" id="ulang">Ulangi quiz</button></div>
    <p class="note">Hasil ini hanya rekomendasi awal. Cek informasi pendaftaran dan jadwal terbaru langsung ke masing-masing UKM.</p>`;
  $("ulang").onclick = () => { jawab.fill(null); idx = 0; r.classList.add("hidden"); $("quiz").classList.remove("hidden"); show(); };
  window.scrollTo({ top: 0, behavior: "smooth" });
}
